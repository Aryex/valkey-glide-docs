GLIDE supports a **custom address resolver**, which allows you to provide a callback that transforms server addresses before connection. The resolver is invoked for each configured address during connection establishment and during cluster topology refreshes.

This is useful for service discovery, proxy setups, and dynamic endpoint resolution in cloud environments.

## Configuration

Add an address resolver to your client configuration. The resolver receives a host and port, and must return a resolved address pair.

<Tabs syncKey="progLangInExamples">
  <TabItem label="Node">
    ```typescript
    import { GlideClient, GlideClusterClient } from "@valkey/valkey-glide";

    const client = await GlideClient.createClient({
        addresses: [{ host: "internal-valkey.service", port: 6379 }],
        addressResolver: (host: string, port: number): [string, number] => {
            // Resolve to a different address
            return ["actual-host.example.com", 6380];
        },
    });
    ```
  </TabItem>

  <TabItem label="Go">
    ```go
    import (
        glide "github.com/valkey-io/valkey-glide/go/v2"
        "github.com/valkey-io/valkey-glide/go/v2/config"
    )

    clientConfig := config.NewClientConfiguration().
        WithAddress(&config.NodeAddress{Host: "internal-valkey.service", Port: 6379}).
        WithAddressResolver(func(host string, port int) (string, int) {
            // Resolve to a different address
            return "actual-host.example.com", 6380
        })

    client, err := glide.NewClient(clientConfig)
    ```
  </TabItem>
</Tabs>

## Use Cases

### Proxy Address Translation

Translate internal cluster addresses to proxy endpoints when connecting through a network proxy:

<Tabs syncKey="progLangInExamples">
  <TabItem label="Node">
    ```typescript
    const PROXY_MAP: Record<string, [string, number]> = {
        "node-1.internal:6379": ["proxy.example.com", 7001],
        "node-2.internal:6379": ["proxy.example.com", 7002],
        "node-3.internal:6379": ["proxy.example.com", 7003],
    };

    const client = await GlideClusterClient.createClient({
        addresses: [{ host: "proxy.example.com", port: 7001 }],
        addressResolver: (host, port) => {
            return PROXY_MAP[`${host}:${port}`] ?? [host, port];
        },
    });
    ```
  </TabItem>

  <TabItem label="Go">
    ```go
    import (
        "fmt"

        glide "github.com/valkey-io/valkey-glide/go/v2"
        "github.com/valkey-io/valkey-glide/go/v2/config"
    )

    type addr struct {
        host string
        port int
    }

    proxyMap := map[string]addr{
        "node-1.internal:6379": {"proxy.example.com", 7001},
        "node-2.internal:6379": {"proxy.example.com", 7002},
        "node-3.internal:6379": {"proxy.example.com", 7003},
    }

    resolver := func(host string, port int) (string, int) {
        key := fmt.Sprintf("%s:%d", host, port)
        if entry, ok := proxyMap[key]; ok {
            return entry.host, entry.port
        }
        return host, port
    }

    clusterConfig := config.NewClusterClientConfiguration().
        WithAddress(&config.NodeAddress{Host: "proxy.example.com", Port: 7001}).
        WithAddressResolver(resolver)

    client, err := glide.NewClusterClient(clusterConfig)
    ```
  </TabItem>
</Tabs>

### Service Discovery

Resolve addresses using a local service registry or cached DNS lookup:

<Tabs syncKey="progLangInExamples">
  <TabItem label="Node">
    ```typescript
    import { lookupServiceSync } from "./service-registry";

    const client = await GlideClient.createClient({
        addresses: [{ host: "valkey-primary", port: 6379 }],
        addressResolver: (host, port) => {
            const resolved = lookupServiceSync(host);
            return [resolved.ip, resolved.port];
        },
    });
    ```
  </TabItem>

  <TabItem label="Go">
    ```go
    import (
        glide "github.com/valkey-io/valkey-glide/go/v2"
        "github.com/valkey-io/valkey-glide/go/v2/config"

        "example.com/myapp/serviceregistry"
    )

    resolver := func(host string, port int) (string, int) {
        resolved := serviceregistry.Lookup(host)
        return resolved.IP, resolved.Port
    }

    clientConfig := config.NewClientConfiguration().
        WithAddress(&config.NodeAddress{Host: "valkey-primary", Port: 6379}).
        WithAddressResolver(resolver)

    client, err := glide.NewClient(clientConfig)
    ```
  </TabItem>
</Tabs>

### Cloud Environment Endpoint Resolution

Map private IPs returned by cluster topology to public endpoints:

<Tabs syncKey="progLangInExamples">
  <TabItem label="Node">
    ```typescript
    const client = await GlideClusterClient.createClient({
        addresses: [{ host: "cluster.cloud-provider.com", port: 6379 }],
        addressResolver: (host, port) => {
            // Replace private VPC IPs with the public endpoint
            if (host.startsWith("10.0.")) {
                return ["cluster.cloud-provider.com", port];
            }
            return [host, port];
        },
    });
    ```
  </TabItem>

  <TabItem label="Go">
    ```go
    import (
        "strings"

        glide "github.com/valkey-io/valkey-glide/go/v2"
        "github.com/valkey-io/valkey-glide/go/v2/config"
    )

    resolver := func(host string, port int) (string, int) {
        // Replace private VPC IPs with the public endpoint
        if strings.HasPrefix(host, "10.0.") {
            return "cluster.cloud-provider.com", port
        }
        return host, port
    }

    clusterConfig := config.NewClusterClientConfiguration().
        WithAddress(&config.NodeAddress{Host: "cluster.cloud-provider.com", Port: 6379}).
        WithAddressResolver(resolver)

    client, err := glide.NewClusterClient(clusterConfig)
    ```
  </TabItem>
</Tabs>

## Behavior and Limitations

### Common

- **Synchronous only**: The resolver must be synchronous in both languages. In Node.js, async resolvers (returning a Promise) are not supported. In Go, the resolver function blocks until it returns.
- **Fallback on failure**: If the resolver throws an exception (Node.js) or panics (Go), or returns an invalid value, the original address is used.
- **Automatic cleanup**: The resolver is automatically removed from the internal registry when the client is closed or connection fails.

### Node.js

- **Main thread execution**: The resolver runs on the Node.js main thread. Avoid long-running logic that could block the event loop.

### Go

- **Concurrency safety**: The resolver function must be safe for concurrent use — it may be called from multiple goroutines simultaneously.
- **Panic recovery**: If the resolver panics or returns an invalid value (empty host, or port outside the 1–65535 range), the original address is used as a fallback.
- **Global resolver registry**: Internally, resolvers are stored in a global registry. If multiple clients use different resolvers, the last one registered takes effect (last-one-wins behavior).
